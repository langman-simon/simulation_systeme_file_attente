import math

MAX_QUEUE = 256
SIMULATION_MINUTES = 20

# ---------- RNG (LCG) ----------
class LCG:
    def __init__(self, x0, a, c, m):
        self.x = x0 % m
        self.a = a
        self.c = c
        self.m = m

    def next_int(self):
        self.x = (self.a * self.x + self.c) % self.m
        return self.x

    def rand_real(self):
        u = self.next_int() / self.m
        if u <= 0.0:
            return 1e-6
        if u >= 1.0:
            return 1.0 - 1e-6
        return u

# ---------- Poisson ----------
def generate_poisson(rng, lam):
    if lam <= 0:
        return 0
    L = math.exp(-lam)
    p = 1.0
    k = 0
    while p > L:
        p *= rng.rand_real()
        k += 1
        if k > 1_000_000:
            break
    return k - 1

# ---------- Service duration ----------
def generate_service_duration(rng):
    u = rng.rand_real()
    thresholds = [
        24/60,
        (24+18)/60,
        (24+18+10)/60,
        (24+18+10+3)/60,
        (24+18+10+3+3)/60
    ]
    for i, t in enumerate(thresholds, start=1):
        if u < t:
            return i
    return 6

# ---------- Types ----------
CLIENT_ORDINAIRE = 0
CLIENT_PRIO_REL = 1
CLIENT_PRIO_ABS = 2

class Client:
    def __init__(self, t, d):
        self.type = t
        self.duration = d

class Station:
    def __init__(self):
        self.occupied = False
        self.remaining = 0
        self.client_type = CLIENT_ORDINAIRE

class ClientQueue:
    def __init__(self):
        self.buf = []
    def enqueue(self, c):
        if len(self.buf) < MAX_QUEUE:
            self.buf.append(c)
    def dequeue(self):
        return self.buf.pop(0)
    def empty(self):
        return not self.buf

# ---------- Costs ----------
def cost_presence_per_hour(t):
    return [15.0, 35.0, 45.0][t]

def cost_occupation_per_hour(t):
    return 28.0 if t == CLIENT_ORDINAIRE else 33.0

def compute_nbStationsMax():
    duree_moy = (1*24 + 2*18 + 3*10 + 4*3 + 5*3 + 6*2) / 60
    charge = 1.5 + 0.7
    return math.ceil(charge * duree_moy * 5)

# ---------- Simulation ----------
def simulate_system(rng):
    best_cost = float("inf")
    best_nb = 1

    for nb in range(1, compute_nbStationsMax() + 1):
        stations = [Station() for _ in range(nb)]
        qOrd, qAbs, qRel = ClientQueue(), ClientQueue(), ClientQueue()
        total_cost = 0.0

        for _ in range(SIMULATION_MINUTES):
            for s in stations:
                if s.occupied:
                    s.remaining -= 1
                    if s.remaining <= 0:
                        s.occupied = False

            for _ in range(generate_poisson(rng, 1.5)):
                qOrd.enqueue(Client(CLIENT_ORDINAIRE, generate_service_duration(rng)))

            for _ in range(generate_poisson(rng, 0.7)):
                t = CLIENT_PRIO_ABS if rng.rand_real() <= 0.3 else CLIENT_PRIO_REL
                (qAbs if t == CLIENT_PRIO_ABS else qRel).enqueue(
                    Client(t, generate_service_duration(rng))
                )

            for s in stations:
                if not s.occupied:
                    if not qAbs.empty():
                        c = qAbs.dequeue()
                    elif not qRel.empty():
                        c = qRel.dequeue()
                    elif not qOrd.empty():
                        c = qOrd.dequeue()
                    else:
                        continue
                    s.occupied = True
                    s.remaining = c.duration
                    s.client_type = c.type

            cost_min = 0.0
            for q in (qOrd, qAbs, qRel):
                for c in q.buf:
                    cost_min += cost_presence_per_hour(c.type) / 60

            for s in stations:
                cost_min += (cost_occupation_per_hour(s.client_type) if s.occupied else 18.0) / 60

            total_cost += cost_min

        if total_cost < best_cost:
            best_cost = total_cost
            best_nb = nb

    return best_nb

# ---------- Main ----------
if __name__ == "__main__":
    rng = LCG(2789, 110145, 1, 1 << 31)
    print("Estimated optimal number of stations:", simulate_system(rng))
